// шифр цезаря
const abc = 'абвгдеёжзийклмнопрстуфхцчшщъыьэюя0123456789 ,.?'
const n = 47

function go() {
    let t = document.getElementById('txt').value
    let s = parseInt(document.getElementById('sh').value) || 0
    let r = ''
    for (let i = 0; i < t.length; i++) {
        let p = abc.indexOf(t[i])
        if (p == -1) {
            r += t[i]
        } else {
            r += abc[(p + s + n) % n]
        }
    }
    document.getElementById('res').value = r
}

document.querySelector('#create').onclick = go;

var koi8Inv = {}
for (let ch in koi8) {
    koi8Inv[koi8[ch]] = ch
}

function modInv(a, m) {
    var m0 = m
    var x0 = 0
    var x1 = 1
    if (m == 1) return 0
    while (a > 1) {
        let qq = Math.floor(a / m)
        let t = m
        m = a % m
        a = t
        t = x0
        x0 = x1 - qq * x0
        x1 = t
    }
    if (x1 < 0) x1 += m0
    return x1
}

function go() {
    var q = parseInt(document.getElementById('q').value)
    var r = parseInt(document.getElementById('r').value)
    var wStr = document.getElementById('w').value
    var enc = document.getElementById('enc').value

    var W = wStr.split(',').map(Number)
    var rInv = modInv(r, q)
    var B = []
    for (let i = 0; i < W.length; i++) {
        B[i] = (W[i] * r) % q
    }

    var rStr = ''
    for (let k = 0; k < enc.length; k++) {
        let c = enc.charCodeAt(k)
        let cp = (c * rInv) % q
        let code = 0
        for (let i = W.length - 1; i >= 0; i--) {
            if (cp >= W[i]) {
                code = code | (1 << (W.length - 1 - i))
                cp = cp - W[i]
            }
        }
        if (code > 127 && koi8Inv[code]) {
            rStr += koi8Inv[code]
        } else {
            rStr += String.fromCharCode(code)
        }
    }
    document.getElementById('res').textContent = rStr
}

document.querySelector('#create').onclick = go;
